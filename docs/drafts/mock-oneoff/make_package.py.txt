"""Build the mock extraction package: ED readmissions 2021-2024 (fictional)."""
import csv, json, random, sys
from datetime import date, timedelta
from pathlib import Path

out = Path(sys.argv[1]); out.mkdir(parents=True, exist_ok=True)
rng = random.Random(20261122)
START, END = date(2021, 1, 1), date(2024, 12, 31)
HOSP = ["H01", "H02", "H03", "H04", "H05"]

def rdate(a, b):
    return a + timedelta(days=rng.randint(0, (b - a).days))

people = [f"P{100000 + i}" for i in range(160)]
ed = []
for i in range(240):
    ed.append({"presentation_id": f"ED-{2100000 + i}", "person_key": rng.choice(people),
               "presentation_date": rdate(START, END).isoformat(),
               "triage_category": rng.choices([1, 2, 3, 4, 5], [2, 12, 35, 38, 13])[0],
               "hospital_code": rng.choice(HOSP)})
ed.sort(key=lambda r: r["presentation_date"])
# PLANTED 1: a null identifier (row 137's presentation_id left empty)
ed[137]["presentation_id"] = ""

adm = []
for i, e in enumerate(rng.sample(ed, 96)):
    d0 = date.fromisoformat(e["presentation_date"]) + timedelta(days=rng.randint(0, 45))
    adm.append({"admission_id": f"AD-{5300000 + i}", "person_key": e["person_key"],
                "admission_date": min(d0, END).isoformat(),
                "length_of_stay_days": rng.choices(range(0, 15), [5]+[9]*6+[3]*8)[0],
                "within_30_days": str((d0 - date.fromisoformat(e['presentation_date'])).days <= 30).lower()})
adm.sort(key=lambda r: r["admission_date"])
# PLANTED 2: one admission dated outside the cohort window (Feb 2025)
adm[-1]["admission_date"] = "2025-02-14"

deaths = []
for i, p in enumerate(rng.sample(people, 11)):
    deaths.append({"death_registration_id": f"DR-{880000 + i}", "person_key": p,
                   "date_of_death": rdate(date(2021, 3, 1), date(2025, 6, 30)).isoformat(),
                   "cause_group": rng.choice(["circulatory", "neoplasm", "respiratory", "external", "other"])})
deaths.sort(key=lambda r: r["date_of_death"])

summ = []
for y in range(2021, 2025):
    for q in range(1, 5):
        pres = sum(1 for e in ed if e["presentation_date"][:4] == str(y) and (int(e["presentation_date"][5:7]) - 1) // 3 + 1 == q)
        re_ = sum(1 for a in adm if a["admission_date"][:4] == str(y) and (int(a["admission_date"][5:7]) - 1) // 3 + 1 == q and a["within_30_days"] == "true")
        summ.append({"year": y, "quarter": q, "presentations": pres, "readmissions_30d": re_,
                     "readmission_rate": f"{(re_ / pres if pres else 0):.3f}"})
# PLANTED 3: a wrongly typed value in the derived table (a suppressed cell written as text)
summ[6]["readmission_rate"] = "suppressed"

def write(name, rows):
    with open(out / name, "w", newline="") as f:
        w = csv.DictWriter(f, fieldnames=list(rows[0])); w.writeheader(); w.writerows(rows)

write("ed_presentations.csv", ed); write("hospital_admissions.csv", adm)
write("death_registrations.csv", deaths); write("readmission_summary.csv", summ)
# PLANTED 4: an undeclared extra file - the analyst's draft linkage keys, sent by mistake
write("linkage_keys_DRAFT.csv", [{"person_key": p, "linkage_key": f"LK{rng.randint(10**8, 10**9)}"} for p in people[:25]])

H = {"id": "hospital-services", "name": "Hospital Services Agency (mock)"}
R = {"id": "civil-registry", "name": "Civil Registry Office (mock)"}
def col(n, t, pi=False): return {"name": n, "type": t, "primary_identifier": pi}
recipe = {
  "recipe_format_version": 1,
  "extraction_id": "ed-readmissions-2021-2024",
  "project": "RX-0417 Readmission after an emergency department visit, 2021-2024 (fictional study, Example University)",
  "filter": "Every emergency department presentation between 1 January 2021 and 31 December 2024 at the five study hospitals; each presenting person's inpatient admissions in the same window; and any death registration for those people up to 30 June 2025.",
  "tables": [
    {"file": "ed_presentations.csv", "format": "csv", "agency": H,
     "collection": {"id": "emergency-care", "name": "Emergency Care"},
     "dataset": {"id": "ed-presentations", "name": "ED presentations", "table": "ed_presentations"},
     "derived": False,
     "columns": [col("presentation_id", "string", True), col("person_key", "string"),
                 col("presentation_date", "date"), col("triage_category", "integer"),
                 col("hospital_code", "string")],
     "cohort": [{"column": "presentation_date", "lower": "2021-01-01", "upper": "2024-12-31"}],
     "expected_rows": 240},
    {"file": "hospital_admissions.csv", "format": "csv", "agency": H,
     "collection": {"id": "inpatient-care", "name": "Inpatient Care"},
     "dataset": {"id": "hospital-admissions", "name": "Hospital admissions", "table": "hospital_admissions"},
     "derived": False,
     "columns": [col("admission_id", "string", True), col("person_key", "string"),
                 col("admission_date", "date"), col("length_of_stay_days", "integer"),
                 col("within_30_days", "boolean")],
     "cohort": [{"column": "admission_date", "lower": "2021-01-01", "upper": "2024-12-31"}],
     "expected_rows": 96},
    {"file": "readmission_summary.csv", "format": "csv", "agency": H,
     "collection": {"id": "inpatient-care", "name": "Inpatient Care"},
     "dataset": {"id": "readmission-summary", "name": "Quarterly readmission summary", "table": "readmission_summary"},
     "derived": True,
     "columns": [col("year", "integer", True), col("quarter", "integer", True),
                 col("presentations", "integer"), col("readmissions_30d", "integer"),
                 col("readmission_rate", "number")],
     "cohort": [{"column": "year", "lower": 2021, "upper": 2024}],
     "expected_rows": 16},
    {"file": "death_registrations.csv", "format": "csv", "agency": R,
     "collection": {"id": "death-registrations", "name": "Death Registrations"},
     "dataset": {"id": "deaths", "name": "Death registrations", "table": "death_registrations"},
     "derived": False,
     "columns": [col("death_registration_id", "string", True), col("person_key", "string"),
                 col("date_of_death", "date"), col("cause_group", "string")],
     "cohort": [{"column": "date_of_death", "lower": "2021-01-01", "upper": "2025-06-30"}],
     "expected_rows": 11},
  ]}
(out / "recipe.json").write_text(json.dumps(recipe, indent=2) + "\n")
print("wrote", sorted(p.name for p in out.iterdir()))
